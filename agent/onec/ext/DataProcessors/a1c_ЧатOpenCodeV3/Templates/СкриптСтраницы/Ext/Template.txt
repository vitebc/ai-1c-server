/* Chat page functions. 1C calls them through
   Elements.<HTMLField>.Document.defaultView.<name>(...).
   They keep no state in variables: every call carries ids and full HTML.
   The page is rendered once; afterwards only blocks are replaced, so the
   scroll position and the already rendered messages are kept. */

function byId(id) {
  return document.getElementById(id);
}

function isAtBottom() {
  var height = document.body.scrollHeight;
  return (window.pageYOffset + window.innerHeight) >= (height - 80);
}

function toBottom() {
  try {
    window.scrollTo(0, document.body.scrollHeight);
  } catch (e) {
  }
}

/* html is a complete <div class="msg ..." id="..."> block. */
function upsertMessage(id, html) {
  var box = byId('feed');
  if (!box) {
    return;
  }
  var wasBottom = isAtBottom();
  var old = byId(id);
  var holder = document.createElement('div');
  holder.innerHTML = String(html).replace(/^\s+/, '');
  var node = holder.firstChild;
  if (!node) {
    return;
  }
  if (old) {
    box.replaceChild(node, old);
  } else {
    box.appendChild(node);
  }
  if (wasBottom) {
    toBottom();
  }
}

function setMeta(id, text) {
  var msg = byId(id);
  if (!msg) {
    return;
  }
  var wasBottom = isAtBottom();
  var meta = msg.getElementsByClassName ? msg.getElementsByClassName('meta')[0] : null;
  if (!meta) {
    meta = document.createElement('div');
    meta.className = 'meta';
    msg.appendChild(meta);
  }
  meta.textContent = text;
  if (wasBottom) {
    toBottom();
  }
}

function showQuestion(html) {
  var box = byId('question');
  if (!box) {
    return;
  }
  box.innerHTML = html;
  box.style.display = 'block';
  toBottom();
}

function hideQuestion() {
  var box = byId('question');
  if (!box) {
    return;
  }
  box.innerHTML = '';
  box.style.display = 'none';
}

/* Highlights the chosen option of one field without rebuilding the card, so
   text typed into other fields is not lost. */
function selectOption(field, opt) {
  var box = byId('opts_' + field);
  if (!box) {
    return;
  }
  var links = box.getElementsByTagName('a');
  for (var i = 0; i < links.length; i++) {
    links[i].className = (i === opt) ? 'opt on' : 'opt';
  }
}

/* Switches the visible question tab. No state is kept: the caller sends the
   full card HTML again, so this only matters for the click before the reply. */
function showTab(idx) {
  var tabs = document.getElementsByClassName('q-tab');
  var panels = document.getElementsByClassName('q-panel');
  for (var i = 0; i < tabs.length; i++) {
    tabs[i].className = (i === idx) ? 'q-tab on' : 'q-tab';
  }
  for (var j = 0; j < panels.length; j++) {
    panels[j].className = (j === idx) ? 'q-panel on' : 'q-panel';
  }
}

/* Removes the "not answered" dot of one tab. */
function clearTabDot(idx) {
  var tabs = document.getElementsByClassName('q-tab');
  if (idx < 0 || idx >= tabs.length) {
    return;
  }
  var dots = tabs[idx].getElementsByClassName('q-dot');
  while (dots.length > 0) {
    dots[0].parentNode.removeChild(dots[0]);
  }
}

/* Updates the "N left" counter under the answer button. Texts are passed by
   the caller: the page templates stay pure ASCII, Cyrillic comes from BSL. */
function setLeftCount(n, textWarn, textDone) {
  var el = byId('qleft');
  if (!el) {
    return;
  }
  el.textContent = (n > 0) ? textWarn : textDone;
  el.className = (n > 0) ? 'status warn' : 'status';
}

function setValue(id, value) {
  var el = byId(id);
  if (el) {
    el.value = value;
  }
}

function setStatus(text, kind) {
  var s = byId('status');
  if (!s) {
    return;
  }
  if (!text) {
    s.textContent = '';
    s.style.display = 'none';
    return;
  }
  s.textContent = text;
  s.className = 'status ' + (kind || '');
  s.style.display = 'inline-block';
}

function setBusy(on) {
  document.body.className = on ? 'busy' : '';
}

function setHeader(html) {
  var h = byId('hdr');
  if (h) {
    h.innerHTML = html;
  }
}

function clearFeed() {
  var f = byId('feed');
  if (f) {
    f.innerHTML = '';
  }
}

function clearPrompt() {
  var p = byId('prompt');
  if (p) {
    p.value = '';
  }
  focusPrompt();
}

function focusPrompt() {
  var p = byId('prompt');
  if (!p) {
    return;
  }
  try {
    p.focus();
  } catch (e) {
  }
}

function getValue(id) {
  var e = byId(id);
  return e ? e.value : '';
}

function onPromptKey(e) {
  e = e || window.event;
  if (!e || e.keyCode !== 13 || e.shiftKey) {
    return;
  }
  e.preventDefault();
  var s = byId('send');
  if (s) {
    s.click();
  }
}

/* Enter in the free-answer field answers the question. */
function onQuestionKey(e) {
  e = e || window.event;
  if (!e || e.keyCode !== 13) {
    return;
  }
  if (e.preventDefault) {
    e.preventDefault();
  }
  var a = byId('answer');
  if (a) {
    a.click();
  }
}
