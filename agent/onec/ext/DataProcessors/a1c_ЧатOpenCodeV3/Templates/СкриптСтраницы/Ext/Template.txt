/* Chat page functions. 1C calls them through
   Elements.<HTMLField>.Document.defaultView.<name>(...).
   They keep no state in variables: every call carries ids and full HTML.
   The page is rendered once; afterwards only blocks are replaced, so the
   scroll position and the already rendered messages are kept. */

function byId(id) {
  return document.getElementById(id);
}

function isAtBottom() {
  var height = document.body.scrollHeight;
  return (window.pageYOffset + window.innerHeight) >= (height - 80);
}

function toBottom() {
  try {
    window.scrollTo(0, document.body.scrollHeight);
  } catch (e) {
  }
}

/* html is a complete <div class="msg ..." id="..."> block. */
function upsertMessage(id, html) {
  var box = byId('feed');
  if (!box) {
    return;
  }
  var wasBottom = isAtBottom();
  var old = byId(id);
  var holder = document.createElement('div');
  holder.innerHTML = String(html).replace(/^\s+/, '');
  var node = holder.firstChild;
  if (!node) {
    return;
  }
  if (old) {
    box.replaceChild(node, old);
  } else {
    box.appendChild(node);
  }
  /* The pending bubble always stays at the very end of the feed. */
  var pending = byId('pending');
  if (pending && pending.parentNode === box) {
    box.appendChild(pending);
  }
  if (wasBottom) {
    toBottom();
  }
}

function setMeta(id, text) {
  var msg = byId(id);
  if (!msg) {
    return;
  }
  var wasBottom = isAtBottom();
  var meta = msg.getElementsByClassName ? msg.getElementsByClassName('meta')[0] : null;
  if (!meta) {
    meta = document.createElement('div');
    meta.className = 'meta';
    msg.appendChild(meta);
  }
  meta.textContent = text;
  if (wasBottom) {
    toBottom();
  }
}

/* "Agent is working" bubble: shown right after the user message, replaced by
   the real answer when it arrives. Texts come from BSL (page stays ASCII). */
function showPending(text) {
  var msg = byId('pending');
  if (!msg) {
    return;
  }
  var content = msg.getElementsByClassName('content')[0];
  if (content) {
    content.textContent = text;
  }
  /* Move to the end of the feed: appended messages would otherwise push the
     bubble to the top of the conversation. */
  var feed = byId('feed');
  if (feed) {
    feed.appendChild(msg);
  }
  msg.style.display = 'block';
  toBottom();
}

function hidePending() {
  var msg = byId('pending');
  if (msg) {
    msg.style.display = 'none';
  }
}

function showQuestion(html) {
  var box = byId('question');
  if (!box) {
    return;
  }
  box.innerHTML = html;
  box.style.display = 'block';
  toBottom();
}

function hideQuestion() {
  var box = byId('question');
  if (!box) {
    return;
  }
  box.innerHTML = '';
  box.style.display = 'none';
}

/* Highlights the chosen option of one field without rebuilding the card, so
   text typed into other fields is not lost. */
function selectOption(field, opt) {
  var box = byId('opts_' + field);
  if (!box) {
    return;
  }
  var links = box.getElementsByTagName('a');
  for (var i = 0; i < links.length; i++) {
    links[i].className = (i === opt) ? 'opt on' : 'opt';
  }
}

/* Switches the visible question tab. No state is kept: the caller sends the
   full card HTML again, so this only matters for the click before the reply. */
function showTab(idx) {
  var tabs = document.getElementsByClassName('q-tab');
  var panels = document.getElementsByClassName('q-panel');
  for (var i = 0; i < tabs.length; i++) {
    tabs[i].className = (i === idx) ? 'q-tab on' : 'q-tab';
  }
  for (var j = 0; j < panels.length; j++) {
    panels[j].className = (j === idx) ? 'q-panel on' : 'q-panel';
  }
}

/* Removes the "not answered" dot of one tab. */
function clearTabDot(idx) {
  var tabs = document.getElementsByClassName('q-tab');
  if (idx < 0 || idx >= tabs.length) {
    return;
  }
  var dots = tabs[idx].getElementsByClassName('q-dot');
  while (dots.length > 0) {
    dots[0].parentNode.removeChild(dots[0]);
  }
}

/* Updates the "N left" counter under the answer button. Texts are passed by
   the caller: the page templates stay pure ASCII, Cyrillic comes from BSL. */
function setLeftCount(n, textWarn, textDone) {
  var el = byId('qleft');
  if (!el) {
    return;
  }
  el.textContent = (n > 0) ? textWarn : textDone;
  el.className = (n > 0) ? 'status warn' : 'status';
}

function setValue(id, value) {
  var el = byId(id);
  if (el) {
    el.value = value;
  }
}

function setStatus(text, kind) {
  var s = byId('status');
  if (!s) {
    return;
  }
  if (!text) {
    s.textContent = '';
    s.style.display = 'none';
    return;
  }
  s.textContent = text;
  s.className = 'status ' + (kind || '');
  s.style.display = 'inline-block';
}

function setBusy(on) {
  document.body.className = on ? 'busy' : '';
  if (!on) {
    focusPrompt();
  }
}

/* Composer: the textarea grows with its text (1 line up to ~7 lines) and the
   page bottom padding follows the composer height, so a long input never
   hides the last message behind it. */
function canSend() {
  var box = byId('chips');
  if (box && box.children.length > 0) {
    return true;
  }
  if (readPasted().length > 2) {
    return true;
  }
  var el = byId('prompt');
  return el ? el.value.trim().length > 0 : false;
}


function syncComposer() {
  var box = byId('cbox');
  var p = byId('prompt');
  if (!box || !p) {
    return;
  }
  p.style.height = 'auto';
  p.style.height = Math.max(96, Math.min(p.scrollHeight, 200)) + 'px';
  box.className = canSend() ? 'c-box' : 'c-box empty';
  var c = byId('composer');
  if (c) {
    document.body.style.paddingBottom = (c.offsetHeight + 16) + 'px';
  }
}



function initComposer() {
  syncComposer();
  focusPrompt();
}

function setHeader(html) {
  var h = byId('hdr');
  if (h) {
    h.innerHTML = html;
  }
}

function clearFeed() {
  var f = byId('feed');
  if (f) {
    f.innerHTML = '';
  }
}

function clearPrompt() {
  var p = byId('prompt');
  if (p) {
    p.value = '';
  }
  syncComposer();
  focusPrompt();
}

function focusPrompt() {
  var p = byId('prompt');
  if (!p) {
    return;
  }
  try {
    p.focus();
  } catch (e) {
  }
}

function getValue(id) {
  var e = byId(id);
  return e ? e.value : '';
}

function onPromptKey(e) {
  e = e || window.event;
  if (!e || e.keyCode !== 13 || e.shiftKey) {
    return;
  }
  e.preventDefault();
  // Enter с пустым текстом, но есть вложения — тоже отправляем.
  if (canSend()) {
    var s = byId('send');
    if (s) {
      s.click();
    }
  }
}



/* Enter in the free-answer field answers the question. */
function onQuestionKey(e) {
  e = e || window.event;
  if (!e || e.keyCode !== 13) {
    return;
  }
  if (e.preventDefault) {
    e.preventDefault();
  }
  var a = byId('answer');
  if (a) {
    a.click();
  }
}

/* Attachments. Chips live in #chips (composer), pasted images are kept as a
   JSON array of data-URIs in the hidden #pastebuf field. 1C reads both at
   send time: chips are removed from the DOM, pastebuf is cleared. */

function chipHTML(id, name) {
  return "<a class='chip-x' href='#act:detach|" + id + "' title='" + name + "'>&times;</a>"
    + "<span class='chip-name'>" + name + "</span>";
}

function addChip(id, name) {
  var box = byId('chips');
  if (!box) {
    return;
  }
  var chip = document.createElement('div');
  chip.className = 'chip';
  chip.id = 'chip_' + id;
  chip.innerHTML = chipHTML(id, name);
  box.appendChild(chip);
  renderChips();
}

function removeChip(id) {
  var chip = byId('chip_' + id);
  if (chip && chip.parentNode) {
    chip.parentNode.removeChild(chip);
  }
  renderChips();
}

function clearChips() {
  var box = byId('chips');
  if (box) {
    box.innerHTML = '';
  }
  renderChips();
}

/* Hides the empty placeholder and resizes the composer: chips take space. */
function renderChips() {
  var box = byId('chips');
  if (!box) {
    return;
  }
  box.style.display = (box.children.length > 0) ? 'flex' : 'none';
  syncComposer();
}

/* Ctrl+V in the prompt: images from the clipboard are downscaled to a
   data-URI and appended to the pastebuf JSON array. Non-image pastes keep
   the browser default (text insertion). */
function onPromptPaste(e) {
  e = e || window.event;
  var items = (e.clipboardData && e.clipboardData.items) ? e.clipboardData.items : null;
  if (!items) {
    return;
  }
  var foundImage = false;
  for (var i = 0; i < items.length; i++) {
    if (items[i].type && String(items[i].type).indexOf('image/') === 0) {
      foundImage = true;
      break;
    }
  }
  if (!foundImage) {
    return;
  }
  if (e.preventDefault) {
    e.preventDefault();
  }
  var buf = byId('pastebuf');
  if (!buf) {
    return;
  }
  var list = [];
  try {
    var parsed = JSON.parse(buf.value || '[]');
    if (parsed && parsed.length) {
      list = parsed;
    }
  } catch (err) {
    list = [];
  }
  for (var j = 0; j < items.length; j++) {
    if (!(items[j].type && String(items[j].type).indexOf('image/') === 0)) {
      continue;
    }
    var file = items[j].getAsFile ? items[j].getAsFile() : null;
    if (!file) {
      continue;
    }
    (function(fileItem, mime) {
      var reader = new FileReader();
      reader.onload = function(ev) {
        var img = document.createElement('img');
        img.onload = function() {
          var scale = Math.min(1, 1600 / Math.max(img.width, img.height));
          var w = Math.max(1, Math.round(img.width * scale));
          var h = Math.max(1, Math.round(img.height * scale));
          var canvas = document.createElement('canvas');
          canvas.width = w;
          canvas.height = h;
          try {
            canvas.getContext('2d').drawImage(img, 0, 0, w, h);
            list.push(canvas.toDataURL(mime === 'image/png' ? 'image/png' : 'image/jpeg', 0.85));
          } catch (err) {
            return;
          }
          buf.value = JSON.stringify(list);
          renderPasteChips();
        };
        img.onerror = function() {};
        img.src = ev.target.result;
      };
      reader.readAsDataURL(fileItem);
    })(file, items[j].type);
  }
}

function readPasted() {
  var buf = byId('pastebuf');
  return buf ? (buf.value || '') : '';
}

function clearPasted() {
  var buf = byId('pastebuf');
  if (buf) {
    buf.value = '';
  }
  renderPasteChips();
}

/* Pasted-image chips: one per data-URI in #pastebuf, with a thumbnail. The
   chip row is shared with file chips (#chips); paste chips carry the "p"
   prefix and are managed together with pastebuf (clearPasted removes them). */

function pasteChipHTML(n, dataUri) {
  var thumb = '';
  if (dataUri) {
    thumb = "<img class='chip-thumb' src='" + dataUri + "' alt=''/>";
  }
  return "<a class='chip-x' href='#act:detach|p" + n + "' title='"
    + 'Убрать вставку' + "'>&times;</a>"
    + thumb
    + "<span class='chip-name'>" + 'Вставка ' + (n + 1) + "</span>";
}

function renderPasteChips() {
  var buf = byId('pastebuf');
  var box = byId('chips');
  if (!buf || !box) {
    return;
  }
  // Remove existing paste chips.
  var nodes = box.querySelectorAll('.chip.paste');
  for (var i = 0; i < nodes.length; i++) {
    nodes[i].parentNode.removeChild(nodes[i]);
  }
  var list = [];
  try {
    var parsed = JSON.parse(buf.value || '[]');
    if (parsed && parsed.length) {
      list = parsed;
    }
  } catch (err) {
    list = [];
  }
  for (var j = 0; j < list.length; j++) {
    var chip = document.createElement('div');
    chip.className = 'chip paste';
    chip.id = 'chip_p' + j;
    chip.innerHTML = pasteChipHTML(j, list[j]);
    box.appendChild(chip);
  }
  renderChips();
}

function removePasteChip(n) {
  var buf = byId('pastebuf');
  if (!buf) {
    return;
  }
  var list = [];
  try {
    var parsed = JSON.parse(buf.value || '[]');
    if (parsed && parsed.length) {
      list = parsed;
    }
  } catch (err) {
    list = [];
  }
  if (n >= 0 && n < list.length) {
    list.splice(n, 1);
  }
  buf.value = JSON.stringify(list);
  renderPasteChips();
}
